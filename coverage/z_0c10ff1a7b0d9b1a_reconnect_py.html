40@dataclass(frozen=True, slots=True) 

41class ReconnectPolicy: 

42 """ 

43 Exponential backoff between reconnect attempts: the first attempt is made 

44 after initial_delay, every next delay is multiplied by multiplier up to 

45 max_delay. Each delay is reduced by a random part up to jitter (0.5 means 

46 up to 50%), so many clients don't reconnect to a restarted server at the 

47 same moment. 

48 

49 Times are in seconds. 

50 """ 

51 

52 initial_delay: float = 1.0 

53 max_delay: float = 60.0 

54 multiplier: float = 2.0 

55 jitter: float = 0.5 

56 # None means the client never gives up 

57 max_attempts: Optional[int] = None 

58 # time for TCP/TLS connect and CONNACK of one attempt 

59 connect_timeout: float = 10.0 

60 

61 def __post_init__(self) -> None: 

62 if self.initial_delay < 0 or self.max_delay < self.initial_delay: 

63 raise ValueError("Expected 0 <= initial_delay <= max_delay") 

64 

65 if self.multiplier < 1: 

66 raise ValueError("Expected multiplier >= 1") 

67 

68 if not 0 <= self.jitter <= 1: 

69 raise ValueError("Expected 0 <= jitter <= 1") 

70 

71 if self.max_attempts is not None and self.max_attempts < 1: 

72 raise ValueError("Expected max_attempts >= 1 or None") 

73 

74 if self.connect_timeout <= 0: 

75 raise ValueError("Expected connect_timeout > 0") 

76 

77 def delays(self) -> Iterator[float]: 

78 """Delays before each attempt.""" 

79 delay = self.initial_delay 

80 attempt = 0 

81 

82 while self.max_attempts is None or attempt < self.max_attempts: 

83 yield delay * (1 - self.jitter * random.random()) 

84 

85 attempt += 1 

86 delay = min(delay * self.multiplier, self.max_delay) 

87
